"""
core/state.py -- State vector and world transition
S = (R, E, G, O, K, X) in [0, 130]

v2 (Apr 22, 2026): Transition coefficients aligned with v5.2 baseline
(baseline_v52/nrmo_omega_full/core/state.py). Previous form diverged
from monograph canonical dynamics; that divergence was invisible until
v5.2 baseline became available for side-by-side comparison.
"""
import math
from config.defaults import *


def clip(val, lo=0, hi=130):
    return max(lo, min(hi, val))


def _exp_sample(mean, rng):
    """Exponential(mean) via inverse CDF using uniform rng()."""
    u = max(1e-9, rng())
    return -mean * math.log(u)


def _stdnormal(rng):
    """Box-Muller standard normal from two uniforms."""
    u1 = max(1e-9, rng())
    u2 = rng()
    return math.sqrt(-2 * math.log(u1)) * math.cos(2 * math.pi * u2)


def transition(state, action, world_params, rng):
    """
    State transition function (v5.2-aligned).
    action = [g, s, l, d] with sum=1, min=0.05
    Returns new state.
    """
    R, E, G, O, K, X = state
    g, sf, lr, di = action
    wp = world_params

    # Deterministic dynamics (v5.2 coefficients)
    dR = g * (4.5 + 0.04 * K) - wp["environmental_drag"] * R * 0.18 \
         - g * R * 0.006 * (1 + wp["rivalry_level"]) + 0.008 * (60 - R)
    dE = sf * 4.2 + di * 1.4 - g * 2.3 * (1 + wp["rivalry_level"] * 0.5) + 0.006 * (65 - E)
    dG = di * 4.8 + sf * 1.8 - wp["governance_drag"] * (1.2 + g * 1.5) \
         - g * wp["rivalry_level"] * 0.5 + 0.005 * (55 - G)
    dO = lr * 4.2 + 0.025 * K + di * 0.9 - (X / 130) * 1.6 \
         - wp["stagnation_drag"] * 5.5 + 0.005 * (50 - O)
    dK = lr * 4.5 * max(0.5, wp.get("substitutability", 0.5)) - wp["stagnation_drag"] * 2.2 \
         + 0.012 * G * lr + 0.004 * (50 - K)
    dX = g * 4.8 * (1 + wp["rivalry_level"]) - sf * 5.2 - di * 1.3 - 0.04 * X

    # Shock event
    shock = [0.0] * 6
    if rng() < wp["shock_probability"]:
        mag = _exp_sample(wp["shock_scale"], rng)
        vals = [R, E, G, O, K]
        weights = [math.exp(-v / 30) for v in vals]
        total = sum(weights)
        probs = [w / total for w in weights]
        r = rng()
        acc = 0.0
        idx = 0
        for i, p in enumerate(probs):
            acc += p
            if r <= acc:
                idx = i
                break
        shock[idx] -= mag
        shock[5] += mag * 0.35

    # Tail event -- governance G buffers hit on R
    if rng() < wp["tail_probability"]:
        tail = _exp_sample(wp["tail_scale"], rng)
        misspec = wp.get("tail_model_misspecification", 0.03)
        tail *= max(0, 1 + misspec * _stdnormal(rng))
        buf = 0.4 + 0.6 * (G / 130)
        shock[0] -= tail * 0.45 / buf
        shock[1] -= tail * 0.35
        shock[2] -= tail * 0.25
        shock[3] -= tail * 0.20
        shock[5] += tail * 0.55

    deltas = [dR, dE, dG, dO, dK, dX]
    raw = [R, E, G, O, K, X]
    new_state = [clip(raw[i] + deltas[i] + shock[i]) for i in range(6)]
    return new_state


def is_true_ruin(state):
    R, E, G, O, K, X = state
    return R < 8 or E < 8 or G < 8 or O < 6 or X > 92


def ruin_cause(state):
    R, E, G, O, K, X = state
    if X > 92: return "exposure_cascade"
    if E < 8:  return "environment_collapse"
    if R < 8:  return "resource_collapse"
    if G < 8:  return "governance_failure"
    if O < 6:  return "optionality_exhaustion"
    return "unknown"


def survival_margin(state):
    """Distance from ruin conditions -- proxy indicator (no governance values used)."""
    R, E, G, O, K, X = state
    margins = [R - 8, E - 8, G - 8, O - 6, 92 - X]
    return min(margins)


def normalize_action(weights):
    """Normalize action weights to sum=1 with min=0.05."""
    w = [max(0.05, x) for x in weights]
    total = sum(w)
    return [x / total for x in w]

"""
governance/nrmo_core.py -- NRMO Kernel
Constitutional boundary. Veto only. Never touches execution scoring.
DOES NOT define ruin in execution layer.
DOES NOT receive intermediate evaluation scores.
DOES NOT know world name or signal values.
"""
from config.defaults import *


class NRMOKernel:
    """
    The kernel. Every step, unconditional full audit.
    No exceptions. No fast-path. No skip.
    Robustness cannot be traded for speed.
    """

    def __init__(self, tuning=None):
        self.tuning = tuning or {}

    def construct_admissible_set(self, candidates, state):
        """
        Filter candidates through constitutional veto.
        Returns admissible set -- execution searches within this only.
        Formal: A_t = NRMO(X_t)
        """
        admissible = []
        for c in candidates:
            if self._veto(c, state):
                admissible.append(c)
        # If nothing passes, inject minimal-safe candidate
        if not admissible:
            admissible = [self._safe_fallback(state)]
        return admissible

    def _veto(self, action, state):
        """
        Returns True if action is ADMISSIBLE (passes veto).
        Returns False if action is REJECTED.
        """
        R, E, G, O, K, X = state
        g, s, l, d = action

        # Get tunable thresholds
        growth_cap = self.tuning.get("growth_cap", NRMO_GROWTH_CAP)
        exploration_floor = self.tuning.get("exploration_floor", NRMO_EXPLORATION_FLOOR)
        gov_repair_floor = self.tuning.get("gov_repair_floor", NRMO_GOV_REPAIR_FLOOR)
        eco_growth_cap = self.tuning.get("eco_growth_cap", NRMO_ECO_GROWTH_CAP)

        # Unconditional growth cap
        if g > growth_cap:
            return False

        # High exposure -> growth restriction
        if X > NRMO_HIGH_EXPOSURE_THRESHOLD and g > NRMO_HIGH_EXPOSURE_GROWTH_CAP:
            return False

        # Low environment -> growth restriction
        if E < NRMO_LOW_ENV_THRESHOLD and g > NRMO_LOW_ENV_GROWTH_CAP:
            return False

        # Low governance -> minimum distribution required
        if G < NRMO_LOW_GOV_THRESHOLD and d < NRMO_LOW_GOV_MIN_DIST:
            return False

        # Exploration floor
        if l < exploration_floor:
            return False

        # Governance repair floor
        if G < NRMO_GOV_REPAIR_FLOOR_THRESHOLD and d < gov_repair_floor:
            return False

        # Eco growth cap
        if E < NRMO_ECO_GROWTH_CAP_THRESHOLD and g > eco_growth_cap:
            return False

        # Low environment -> minimum safeguard required (prevents E free-fall)
        if E < 35 and s < 0.30:
            return False
        if E < 25 and s < 0.40:
            return False
        if E < 15 and s < 0.50:
            return False

        return True

    def _safe_fallback(self, state):
        """Minimal safe action when nothing passes veto."""
        from core.state import normalize_action
        return normalize_action([0.10, 0.35, 0.30, 0.25])

    def passive_ruin_check(self, history):
        """
        Detect passive ruin from state history.
        Returns (detected: bool, pattern: str)
        Thresholds: conservative to avoid false positives in early steps.
        """
        if len(history) < PASSIVE_RUIN_O_STEPS:
            return False, None

        # Optionality stagnation: O < 18 for 14 consecutive steps
        recent_o = history[-PASSIVE_RUIN_O_STEPS:]
        if all(s[3] < PASSIVE_RUIN_O_THRESHOLD for s in recent_o):
            return True, "passive_optionality_stagnation"

        # Knowledge stagnation: K < 20 for 18 consecutive steps
        if len(history) >= PASSIVE_RUIN_K_STEPS:
            k_recent = history[-PASSIVE_RUIN_K_STEPS:]
            if all(s[4] < PASSIVE_RUIN_K_THRESHOLD for s in k_recent):
                return True, "passive_knowledge_stagnation"

        # Compound decline: requires ALL of:
        # - O is actually low (< 30) AND declining
        # - G is actually low (< 30) AND declining
        # - K is stagnating (< 25)
        # Over 12 steps -- must be genuinely degraded, not just slightly down
        if len(history) >= PASSIVE_RUIN_COMPOUND_STEPS:
            compound = history[-PASSIVE_RUIN_COMPOUND_STEPS:]
            O_low_and_declining = (compound[-1][3] < 30 and
                                   compound[-1][3] < compound[0][3] - 5)
            G_low_and_declining = (compound[-1][2] < 30 and
                                   compound[-1][2] < compound[0][2] - 5)
            K_stagnating = compound[-1][4] < 25
            if O_low_and_declining and G_low_and_declining and K_stagnating:
                return True, "passive_compound_decline"

        return False, None

"""
governance/tuning_layer.py -- Adaptive Tuning Layer
State-responsive NRMO threshold adjustment.
Does NOT modify ruin boundaries. Adjusts operational thresholds only.
Based on design_spec.py Table 10: World-Dependent Tuning Profiles.
"""
from config.defaults import *


class AdaptiveTuningLayer:
    """
    Adjusts NRMO operational thresholds based on current state.
    Hysteresis prevents rapid oscillation.
    Does NOT touch ruin boundaries (NRMO_RUIN_THRESHOLDS).
    """

    def __init__(self):
        self._hysteresis_steps = 0
        self._hysteresis_min = 6
        self._tightened = False
        self._current_tuning = self._default_tuning()

    def _default_tuning(self):
        return {
            "growth_cap": NRMO_GROWTH_CAP,
            "exploration_floor": NRMO_EXPLORATION_FLOOR,
            "gov_repair_floor": NRMO_GOV_REPAIR_FLOOR,
            "eco_growth_cap": NRMO_ECO_GROWTH_CAP,
        }

    def update(self, state, signal):
        """
        Update tuning based on current state and MAPLayer signal.
        Returns current tuning dict for NRMOKernel.
        """
        R, E, G, O, K, X = state
        mode = signal.get("mode", MODE_NEUTRAL)

        # Hysteresis: don't relax if recently tightened
        if self._hysteresis_steps > 0:
            self._hysteresis_steps -= 1
            return self._current_tuning

        new_tuning = self._default_tuning()
        tightened = False

        # High exposure -> tighten growth
        if X > 48:
            new_tuning["growth_cap"] = min(new_tuning["growth_cap"], 0.36)
            tightened = True

        # Low optionality -> raise exploration floor
        if O < 38:
            new_tuning["exploration_floor"] = max(new_tuning["exploration_floor"], 0.25)
            tightened = True

        # Low governance -> tighten distribution floor
        if G < 34:
            new_tuning["growth_cap"] = min(new_tuning["growth_cap"], 0.32)
            new_tuning["gov_repair_floor"] = max(new_tuning["gov_repair_floor"], 0.25)
            tightened = True

        # Low environment -> tighten eco growth cap (key for PlanetaryStress)
        if E < 45:
            new_tuning["eco_growth_cap"] = min(new_tuning["eco_growth_cap"], 0.28)
            tightened = True
        if E < 35:
            new_tuning["eco_growth_cap"] = min(new_tuning["eco_growth_cap"], 0.20)
            new_tuning["growth_cap"] = min(new_tuning["growth_cap"], 0.25)
            tightened = True
        if E < 25:
            new_tuning["eco_growth_cap"] = min(new_tuning["eco_growth_cap"], 0.12)
            new_tuning["growth_cap"] = min(new_tuning["growth_cap"], 0.15)
            tightened = True

        # Low knowledge -> raise exploration floor
        if K < 34:
            new_tuning["exploration_floor"] = max(new_tuning["exploration_floor"], 0.28)
            tightened = True

        # Mode-based adjustments
        if mode == MODE_BUFFER:
            new_tuning["growth_cap"] = min(new_tuning["growth_cap"], 0.30)
        elif mode == MODE_RETREAT:
            new_tuning["growth_cap"] = min(new_tuning["growth_cap"], 0.15)
            new_tuning["eco_growth_cap"] = min(new_tuning["eco_growth_cap"], 0.12)
        elif mode == MODE_RACE:
            new_tuning["growth_cap"] = min(0.55, NRMO_GROWTH_CAP)

        # Apply hysteresis on tightening
        if tightened and not self._tightened:
            self._hysteresis_steps = self._hysteresis_min

        self._tightened = tightened
        self._current_tuning = new_tuning
        return new_tuning

"""
engine/map_layer.py -- MAPLayer
Shared information base for Shinobi fleet and Omega Full.
Observes, infers, maps. Does NOT define ruin. Does NOT emit world names.
Implements 3D V-Cache: layered history cache (short/mid/long term).
"""
import math
from config.defaults import *


class ThompsonSampler:
    """Independent Thompson Sampling per signal family."""

    def __init__(self):
        self.defensive = [TS_INITIAL_ALPHA, TS_INITIAL_BETA]  # BUFFER/RETREAT
        self.race = [TS_INITIAL_ALPHA, TS_INITIAL_BETA]       # RACE
        self._prev_danger = 0.0
        self._prev_race = 0.0

    def sample_defensive(self, rng):
        a, b = self.defensive
        # Beta distribution sampling via gamma approximation
        x = self._gamma_sample(a, rng)
        y = self._gamma_sample(b, rng)
        return x / (x + y) if (x + y) > 0 else 0.5

    def sample_race(self, rng):
        a, b = self.race
        x = self._gamma_sample(a, rng)
        y = self._gamma_sample(b, rng)
        return x / (x + y) if (x + y) > 0 else 0.5

    def update_defensive(self, survived):
        if survived:
            self.defensive[0] += 1
        else:
            self.defensive[1] += 1

    def update_race(self, survived):
        if survived:
            self.race[0] += 1
        else:
            self.race[1] += 1

    def reset_if_needed(self, danger_conf, race_conf):
        """Reset on sudden pattern change."""
        danger_drop = self._prev_danger - danger_conf
        race_drop = self._prev_race - race_conf
        if danger_drop > TS_RESET_CONFIDENCE_DROP:
            self.defensive = [TS_INITIAL_ALPHA, TS_INITIAL_BETA]
        if race_drop > TS_RESET_CONFIDENCE_DROP:
            self.race = [TS_INITIAL_ALPHA, TS_INITIAL_BETA]
        self._prev_danger = danger_conf
        self._prev_race = race_conf

    @staticmethod
    def _gamma_sample(alpha, rng):
        """Simple gamma sampling via Marsaglia method approximation."""
        if alpha < 1:
            alpha += 1
        d = alpha - 1/3
        c = 1 / math.sqrt(9 * d)
        while True:
            x = rng() * 6 - 3  # normal approx
            v = (1 + c * x) ** 3
            if v > 0 and math.log(rng() + 1e-10) < 0.5 * x * x + d - d * v + d * math.log(v + 1e-10):
                return d * v
            if rng() > 0.9:
                return max(0.01, alpha * rng())


class VCache3D:
    """3D V-Cache: layered history cache inspired by AMD 3D V-Cache."""

    def __init__(self):
        self.short = []   # last MAP_SHORT_WINDOW steps, full resolution
        self.mid = []     # last MAP_MID_WINDOW steps, compressed
        self.long = []    # all history, pattern-compressed

        self._mid_buffer = []
        self._long_buffer = []

    def push(self, state, world_signal):
        """Add new observation to all cache layers."""
        entry = {"state": state[:], "signal": world_signal}

        # Short-term: full resolution
        self.short.append(entry)
        if len(self.short) > MAP_SHORT_WINDOW:
            evicted = self.short.pop(0)
            self._mid_buffer.append(evicted)

        # Mid-term: average compression
        if len(self._mid_buffer) >= 5:
            compressed = self._compress(self._mid_buffer)
            self.mid.append(compressed)
            self._mid_buffer = []
            if len(self.mid) > MAP_MID_WINDOW // 5:
                old = self.mid.pop(0)
                self._long_buffer.append(old)

        # Long-term: pattern compression
        if len(self._long_buffer) >= 20:
            pattern = self._compress(self._long_buffer)
            self.long.append(pattern)
            self._long_buffer = []

    def get_short(self):
        return self.short

    def get_mid(self):
        return self.mid

    def get_long(self):
        return self.long

    def get_all_states(self):
        """Return all cached states for pattern analysis."""
        states = []
        for layer in [self.long, self.mid, self.short]:
            for entry in layer:
                states.append(entry["state"])
        return states

    @staticmethod
    def _compress(entries):
        """Average compression of multiple entries."""
        if not entries:
            return {"state": [0]*6, "signal": {}}
        n = len(entries)
        avg_state = [
            sum(e["state"][i] for e in entries) / n
            for i in range(6)
        ]
        return {"state": avg_state, "signal": entries[-1]["signal"]}

                self._recovery_failures.pop(0)

        self._update_danger_confidence(state)

    def _update_danger_confidence(self, state):
        """Estimate Vulnerable-like danger from observations only."""
        shock_freq = sum(self._shock_events) / max(1, len(self._shock_events))

        if len(self._recovery_failures) > 0:
            recovery_fail_rate = sum(self._recovery_failures) / len(self._recovery_failures)
        else:
            recovery_fail_rate = 0.0

        if len(self._margin_history) > 0:
            margin_proximity = 1.0 - min(1.0, min(self._margin_history) / 50.0)
        else:
            margin_proximity = 0.0

        raw = (
            0.35 * shock_freq +
            0.35 * recovery_fail_rate +
            0.30 * margin_proximity
        )
        # EMA update
        self.danger_confidence = (
            (1 - MAP_EMA_ALPHA) * self.danger_confidence +
            MAP_EMA_ALPHA * raw
        )

    def _update_race_confidence(self, state):
        """Estimate FastExpansionRace-like pressure from observations only.

        v2.1 (Apr 22, 2026): Removed X-level from rivalry_proxy. Previous
        formula `rivalry_proxy = X/92*0.5 + x_ascent*0.5` produced false
        RACE signal in PlanetaryStress (high env_drag accumulates X without
        competitive pressure), causing Omega Full to relax drift penalty and
        drift into exposure_cascade. Race signature is dynamic (X rising

            x_growth = (self._x_history[-1] - self._x_history[0]) / len(self._x_history)
            x_ascent = max(0, x_growth) / 5.0
        else:

        k_ascent = 0.0
        if len(self._margin_history) >= 5 and len(self._x_history) >= 5:
            # Approximate K ascent via state passed in: we only have X history,
            # so use inverse-margin-trend as a coincident indicator of race
            # pressure (margin shrinks when X rises without compensating gains)
            pass  # x_ascent alone is fine for now; K track not cached here

        # Race signature: sustained X ascent (competition-driven growth pressure)
        # Rivalry pressure now comes only from dynamics, not current state level
        rivalry_proxy = min(1.0, x_ascent)
        self._rivalry_pressure = (
            (1 - MAP_EMA_ALPHA) * self._rivalry_pressure +
            MAP_EMA_ALPHA * rivalry_proxy
        )

        raw = (
            0.50 * self._rivalry_pressure +
            0.50 * x_ascent
        )
        self.race_confidence = (
            (1 - MAP_EMA_ALPHA) * self.race_confidence +
            MAP_EMA_ALPHA * raw
        )

    def _update_world_mixture(self):
        """Estimate mixture proportions of world families."""
        d = self.danger_confidence
        r = self.race_confidence
        n = max(0, 1.0 - d - r)

        self.world_mixture["Vulnerable"] = d * 0.6
        self.world_mixture["FastExpansionRace"] = r * 0.6
        self.world_mixture["PlanetaryStress"] = d * 0.3
        self.world_mixture["LateStagnation"] = (1 - r) * 0.2
        self.world_mixture["Normal"] = max(0.05, n * 0.5)

        # Normalize
        total = sum(self.world_mixture.values())
        for k in self.world_mixture:
            self.world_mixture[k] /= total

    def get_signal(self):
        """
        Return abstract signal dict for consumption by engines.
        Never includes world name. Only abstract observational data.
        """
        mode = self._determine_mode()
        return {
            "mode": mode,
            "danger_confidence": self.danger_confidence,
            "race_confidence": self.race_confidence,
            "world_mixture": dict(self.world_mixture),
            "margin_trend": self._margin_trend(),
            "x_ascent_rate": self._x_ascent_rate(),
            "shock_frequency": sum(self._shock_events) / max(1, len(self._shock_events)),
        }

    def _determine_mode(self):
        """Determine binding mode from confidence levels."""
        d = self.danger_confidence
        r = self.race_confidence

        if d >= MAP_DANGER_THRESHOLDS["hard"]:
            return MODE_RETREAT
        if d >= MAP_DANGER_THRESHOLDS["medium"]:
            return MODE_BUFFER
        if d >= MAP_DANGER_THRESHOLDS["soft"]:
            return MODE_PROBE
        if r >= MAP_DANGER_THRESHOLDS["race"]:
            return MODE_RACE
        return MODE_NEUTRAL

        if len(self._margin_history) < 3:
            return 0.0
        recent = self._margin_history[-5:]
        if len(recent) < 2:
            return 0.0
        return (recent[-1] - recent[0]) / len(recent)

class IntermediateEvaluator:
    """
    Trajectory quality evaluation beyond binary survive/ruin.
    Lives in execution layer. Does NOT use governance constants.
    Uses only proxy indicators from observed state trends.
    """

    def score(self, trajectory_states, mode):
        if len(trajectory_states) < 2:
            return 0.0

        s0 = trajectory_states[0]
        sf = trajectory_states[-1]

        from core.state import survival_margin
        margin_start = survival_margin(s0)
        margin_end = survival_margin(sf)
        margin_score = (margin_end - margin_start) / max(1, abs(margin_start) + 10)
        margin_score = max(-1, min(1, margin_score))

        # D2: Exposure growth cost
        x_growth = sf[5] - s0[5]
        exposure_score = -max(0, x_growth) / 20.0

        # D3: Recovery quality (O and K improvement)
        ok_improvement = ((sf[3] - s0[3]) + (sf[4] - s0[4])) / 20.0
        recovery_score = max(-1, min(1, ok_improvement))

        # D5: Degradation cluster (multiple indicators dropping)
        drops = sum(1 for i in range(5) if sf[i] < s0[i] - 3)
        degradation_score = -0.1 * drops

        # D6: Over-conservatism in RACE mode
        conservatism_score = 0.0
        if mode == MODE_RACE:
            growth_taken = s0[0]  # proxy for action taken
            if sf[0] < s0[0] - 2:
                conservatism_score = -0.1

        return (
            0.35 * margin_score +
            0.25 * exposure_score +
            0.20 * recovery_score +
            0.15 * degradation_score +
            0.05 * conservatism_score
        )

        if E < 25:
            safeguard_bonus = s * (25 - E) / 25 * 0.5

        survived_count = 0
        trajectory_states = [state[:]]

        for _ in range(repeats):
            s = state[:]
            episode_score = 0.0
            alive = True

            for step in range(depth):
                s = transition(s, action, world_params, rng)
                if is_true_ruin(s):
                    alive = False
                    episode_score -= 50
                    break

                # v5.1/v5.2: State-adaptive rollout default.
                # v2.3 (Apr 22, 2026): Attempted to purify this to pure
                # riskadjusted_reference(s) (X-only response), as v5.2 does.
                # 500-run validation: FER +15 (57->72), but Vulnerable -8
                # (24->16) -- Vulnerable dies from E/R/G depletion, not X,
                # so X-only rollout response doesn't protect it. Keeping
                # the multi-factor _adaptive_rollout_default which responds
                # to X, rivalry, G, E jointly. The rollout purity hypothesis
                # is incompatible with the Vulnerable gain from v2.1.
                if step > 0:
                    cont = self._adaptive_rollout_default(s, world_params)
                    s = transition(s, cont, world_params, rng)
                    if is_true_ruin(s):
                        alive = False
                        episode_score -= 50
                        break

                trajectory_states.append(s[:])

            if alive:
                survived_count += 1
                R, E, G, O, K, X = s
                g = action[0]

                # Productivity score -- E and G gated (baseline v5.0)
                productivity = g * min(E/55, 1) * min(G/45, 1)

                # v2.1 (Apr 22, 2026): Mode-aware productivity weight.
                # Defensive modes slightly down-weight productivity so that
                # defensive candidates can compete with growth candidates
                # on score. Calibration: small shifts only -- large shifts
                # cause growth collapse in mixed-mode episodes (measured).
                # v2.5 (Apr 25, 2026): PROBE penalty added (was 1.00).
                # Source attribution test on C3 adapter showed Recovery
                # mode patch (mapped from MAPLayer PROBE+BUFFER+RETREAT)
                # was responsible for FER +12pt. PROBE alone is the most
                # frequent defensive mode in FER (~32% of steps) so a
                # PROBE-specific penalty captures most of the gain.
                if mode == MODE_RETREAT:
                    prod_w = 0.70
                elif mode == MODE_BUFFER:
                    prod_w = 0.85
                elif mode == MODE_PROBE:
                    prod_w = 0.92
                elif mode == MODE_RACE:
                    prod_w = 1.10
                else:  # NEUTRAL
                    prod_w = 1.00

                # v2.1: K-starvation penalty in defensive modes only.
                # Vulnerable deaths via passive_knowledge_stagnation came from
                # over-defensive posture starving K growth. Applied only when
                # mode is defensive AND K is actually near the passive-ruin
                # threshold (K<24), so it doesn't bias healthy trajectories.
                k_starve_penalty = 0.0
                if mode in [MODE_BUFFER, MODE_RETREAT] and K < 24:
                    k_starve_penalty = 0.20 * (24 - K) / 24

                # Base score -- stronger E protection
                e_penalty = max(0, (45 - E) / 45) * 0.5  # penalize E depletion
                base = (
                    productivity * prod_w +
                    0.45 * (O/130) +
                    0.20 * (K/130) +
                    0.15 * (G/130) +
                    0.15 * (E/130) -
                    0.40 * (X/130) -
                    e_penalty -
                    k_starve_penalty
                )

                # Drift penalty (lambda_drift)
                g_sust = self._sustainable_growth(s, world_params)
                drift = self._estimate_drift(g, g_sust, s, world_params)
                base -= lambda_drift * drift

                # Fragility penalty
                from core.state import survival_margin
                margin = survival_margin(s)
                if margin < 20:
                    base -= fragility_penalty * (20 - margin) / 20

                episode_score += base

            total_score += episode_score

        # Intermediate evaluation
        intermediate = self.evaluator.score(trajectory_states, mode)
        base_score = total_score / repeats
        survived_ratio = survived_count / repeats

        # v5.1: Softened scaling -- pure multiplicative was too punishing
        # for deeper rollouts in hostile worlds. Now: 0.4 + 0.6*ratio
        # so 50% survival -> 0.7x instead of 0.5x
        survival_signal = 0.4 + 0.6 * survived_ratio

        # Combine: softened survival x base + intermediate quality + safeguard bonus
        final = base_score * survival_signal + 0.15 * intermediate + safeguard_bonus

        return final

    def _risk_adjusted_reference(self, state):
        """
        v5.2-style pure Risk-Adj reference action.
        Used for rollout continuation. Reacts to X only --
        clean X-responsive g reduction without multi-factor noise.
        This is what v5.2 rolled out with at each step, and is
        hypothesized to contribute to its PStress 65% survival.
        """
        R, E, G, O, K, X = state
        g = max(0.08, min(0.48, 0.36 - 0.18 * (X / 130)))
        sf = 0.26 + 0.16 * (X / 130)
        lr = 0.20
        di = max(0.05, 1 - g - sf - lr)
        return normalize_action([g, sf, lr, di])

    def _adaptive_rollout_default(self, state, world_params):
        """
        v5.1/v5.2: State/world-adaptive rollout continuation action.
        Base is [0.24, 0.26, 0.26, 0.24] but adjusts based on:
          - rivalry: reduce growth, increase safety (prevent X spiral)
          - high X: more safety
          - low G: more governance distribution
        This is execution-layer environmental awareness, not governance coupling.
        """
        g, sf, lr, di = 0.24, 0.26, 0.26, 0.24
        R, E, G, O, K, X = state
        rivalry = (world_params or {}).get("rivalry_level", 0.15)

        if rivalry > 0.20:
            adj = min(0.08, (rivalry - 0.20) * 0.20)
            g -= adj
            sf += adj * 0.7
            di += adj * 0.3
        if X > 40:
            x_adj = min(0.06, (X - 40) / 130 * 0.12)
            sf += x_adj
            g = max(0.08, g - x_adj * 0.6)
        if G < 40:
            g_adj = min(0.05, (40 - G) / 130 * 0.10)
            di += g_adj
            g = max(0.08, g - g_adj * 0.5)
        # Low E also needs safety boost
        if E < 40:
            e_adj = min(0.06, (40 - E) / 130 * 0.15)
            sf += e_adj
            g = max(0.08, g - e_adj * 0.6)

        return normalize_action([max(0.05, g), max(0.05, sf), max(0.05, lr), max(0.05, di)])

    def _sustainable_growth(self, state, world_params):
        R, E, G, O, K, X = state
        env_drag = world_params.get("environmental_drag", 0.02)
        g_sust = 0.28 + 0.08*(E/100) + 0.06*(G/100) - 0.10*(X/100) - 0.05*env_drag
        return max(0.15, min(0.40, g_sust))

    def _estimate_drift(self, g, g_sust, state, world_params):
        R, E, G, O, K, X = state
        env_drag = world_params.get("environmental_drag", 0.02)
        tail_prob = world_params.get("tail_probability", 0.03)
        rivalry = world_params.get("rivalry_level", 0.15)
        excess = max(0, g - g_sust)
        env_sensitivity = env_drag + 0.5 * tail_prob
        exposure_factor = 1 + 0.5 * (X/100)
        gov_buffer = 1 - 0.3 * (G/100)
        drift = excess * env_sensitivity * exposure_factor * gov_buffer

        # v5.1: Rivalry tolerance -- in competitive worlds, some growth
        # excess is necessary to survive; reduce drift penalty slightly
        if rivalry > 0.25:
            drift *= max(0.6, 1.0 - rivalry * 0.5)

        return drift

    @staticmethod
    def _in_admissible(candidate, admissible_set):
        """Check if candidate is close enough to any admissible action."""
        for a in admissible_set:
            dist = sum(abs(candidate[i] - a[i]) for i in range(4))
            if dist < 0.15:
                return True
        return False

def lcg_rng(seed):
    """Simple deterministic RNG."""
    state = seed
    def rng():
        nonlocal state
        state = (state * 1664525 + 1013904223) & 0xFFFFFFFF
        return state / 0xFFFFFFFF
    return rng


def run_episode(world_name, horizon, seed, lambda_drift=1.0):
    """
    Run one complete episode with the full stack.
    Returns episode result dict.
    """
    rng = lcg_rng(seed)

    # Initialize world
    world_params = sample_world_params(world_name, rng)
    state = INITIAL_STATE[:]
    history = [state[:]]

    # Initialize full stack
    nrmo = NRMOKernel()
    tuning_layer = AdaptiveTuningLayer()
    map_layer = MAPLayer()
    shinobi_fleet = ShinobiFleet()
    omega = OmegaFull()
    norn_skuld = NornSkuldSystem()

    # Tracking
    true_ruin = False
    passive_ruin = False
    passive_ruin_cause = None
    lifespan = 0
    prev_state = None

    for t in range(horizon):
        lifespan = t + 1

        # -- 1. MAPLayer: continuous observation ------------------
        signal = map_layer.observe(state, prev_state)
        mode = signal.get("mode", MODE_NEUTRAL)

        # -- 2. NRMO kernel: generate candidate pool ---------------
        # Adaptive tuning update (state-responsive thresholds)
        tuning = tuning_layer.update(state, signal)
        nrmo.tuning = tuning

        # Generate raw candidates for NRMO to filter
        raw_candidates = _generate_raw_candidates(state, rng, mode)

        # Constitutional filter -- unconditional every step
        admissible = nrmo.construct_admissible_set(raw_candidates, state)

        # -- 3. Norn/Skuld: task management -----------------------
        norn_skuld.tick(state, signal, len(admissible), shinobi_fleet)

        # -- 4. Shinobi fleet: advance exploration -----------------
        # Provide NRMO admissibility function for Shinobi to use
        def nrmo_admissible_fn(action, s):
            return nrmo._veto(action, s)

        shinobi_aims = shinobi_fleet.advance_explore(
            state, world_params, signal, rng, nrmo_admissible_fn
        )

        # -- 5. Omega Full: full deep search within aiming points ------
        chosen = omega.select_action(
            state, admissible, shinobi_aims, world_params, signal, rng
        )

        # -- 6. Final NRMO constitutional check --------------------
        # Formal: a_t in A_t must always hold
        if not nrmo._veto(chosen, state):
            chosen = admissible[0] if admissible else normalize_action([0.15, 0.35, 0.25, 0.25])

        # -- 7. State transition ------------------------------------
        prev_state = state[:]
        state = transition(state, chosen, world_params, rng)
        history.append(state[:])

        # -- 8. True ruin check -------------------------------------
        if is_true_ruin(state):
            true_ruin = True
            break

        # -- 9. Passive ruin check ----------------------------------
        pr, pr_cause = nrmo.passive_ruin_check(history)
        if pr:
            passive_ruin = True
            passive_ruin_cause = pr_cause
            break

        # -- 10. Thompson Sampling update ---------------------------
        # End of step: update bandit based on whether we survived
        survived_step = not is_true_ruin(state)
        if mode in [MODE_BUFFER, MODE_RETREAT]:
            map_layer.update_thompson(True, survived_step, "defensive")
        elif mode == MODE_RACE:
            map_layer.update_thompson(True, survived_step, "race")

    # Final state metrics
    final = state
    alive = not true_ruin and not passive_ruin

    return {
        "alive": alive,
        "true_ruin": true_ruin,
        "passive_ruin": passive_ruin,
        "ruin_cause": ruin_cause(state) if true_ruin else passive_ruin_cause,
        "lifespan": lifespan,
        "final_state": final,
        "final_R": final[0],
        "final_E": final[1],
        "final_G": final[2],
        "final_O": final[3],
        "final_K": final[4],
        "final_X": final[5],
        "survival_margin_final": survival_margin(final),
        "norn_summary": norn_skuld.get_summary(),
    }


def _generate_raw_candidates(state, rng, mode):
    """Generate raw candidates for NRMO to filter."""
    from engine.shinobi import BASE_TEMPLATES
    candidates = []

    # Base templates
    for t in BASE_TEMPLATES:
        noisy = normalize_action([x + (rng() - 0.5) * 0.06 for x in t])
        candidates.append(noisy)

    # Mode-specific additions
    if mode == MODE_RETREAT:
        candidates.append(normalize_action([0.05, 0.55, 0.20, 0.20]))
        candidates.append(normalize_action([0.05, 0.05, 0.05, 0.85]))
    elif mode == MODE_RACE:
        candidates.append(normalize_action([0.50, 0.20, 0.15, 0.15]))
        candidates.append(normalize_action([0.45, 0.25, 0.15, 0.15]))

    # Risk-adjusted reference
    R, E, G, O, K, X = state
    risk_g = max(0.08, 0.36 - 0.18 * (X/130))
    risk_s = 0.26 + 0.16 * (X/130)
    candidates.append(normalize_action([risk_g, risk_s, 0.20, max(0.05, 1-risk_g-risk_s-0.20)]))

    return candidates


def run_experiment(worlds, strategies, horizon, runs_per_cell):
    """
    Full factorial experiment.
    strategies: list of strategy names (currently only "Omega_Full_Full_Stack")

    """
    results = {}

    for world in worlds:
        results[world] = {}
        for strategy in strategies:
            episodes = []
            for run in range(runs_per_cell):

                ep = run_episode(world, horizon, seed)
                episodes.append(ep)
            results[world][strategy] = episodes

    return results

from config.defaults import *


def aggregate(episodes):
    """Aggregate episode results into summary metrics."""
    n = len(episodes)
    if n == 0:
        return {}

    surv = sum(1 for e in episodes if e["alive"]) / n
    true_ruin = sum(1 for e in episodes if e["true_ruin"]) / n
    passive_ruin = sum(1 for e in episodes if e["passive_ruin"]) / n
    avg_life = sum(e["lifespan"] for e in episodes) / n
    avg_O = sum(e["final_O"] for e in episodes) / n
    avg_X = sum(e["final_X"] for e in episodes) / n
    avg_margin = sum(e["survival_margin_final"] for e in episodes) / n

    score = (
        SCORE_WEIGHTS["surv"] * surv +
        SCORE_WEIGHTS["true_ruin"] * true_ruin +
        SCORE_WEIGHTS["passive_ruin"] * passive_ruin +
        SCORE_WEIGHTS["optionality"] * (avg_O / 130) +
        SCORE_WEIGHTS["exposure"] * (avg_X / 130)
    )

    # Ruin cause breakdown
    causes = {}
    for e in episodes:
        if e["ruin_cause"]:
            causes[e["ruin_cause"]] = causes.get(e["ruin_cause"], 0) + 1

    return {
        "n": n,
        "surv": surv,
        "true_ruin": true_ruin,
        "passive_ruin": passive_ruin,
        "avg_life": avg_life,
        "avg_O": avg_O,
        "avg_X": avg_X,
        "avg_margin": avg_margin,
        "score": score,
        "ruin_causes": causes,
    }
