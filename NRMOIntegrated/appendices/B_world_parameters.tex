\chapter{World Parameter Reference}
\label{app:world-params}

\nrmobox{Document scope}{
This appendix gives the concrete numeric ranges for the five world
families used by the v2.5 codebase, plus the full set of NRMO veto
thresholds and $\Omega$ Full configuration constants.\\[0.3em]
\textbf{Source}: \texttt{nrmo\_full\_v2\_5\_final.zip /
core/worlds.py + config/defaults.py}.
}
