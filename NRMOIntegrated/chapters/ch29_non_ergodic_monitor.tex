% =====================================================================
\section{State Variables}
\label{sec:non-ergodic-state}

\begin{itemize}
\item \texttt{cumulative\_irreversible\_risk}: Cumulative
irreversible risk ($0.0$--$1.0$).
\item \texttt{irreversible\_event\_count}: Number of irreversible
events.
\end{itemize}

% =====================================================================
\section{Trigger}
\label{sec:non-ergodic-trigger}

\begin{equation}
\mathrm{cumulative\_irreversible\_risk} \ge 0.80
\;\Rightarrow\;
\mathrm{SAFE\_EXIT}.
\label{eq:non-ergodic-trigger}
\end{equation}
