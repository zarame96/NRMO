\section{Pass Condition}
\label{sec:test-suite-pass-condition}

\nrmobox{Pass Condition: No violation of Invariants I1--I5}{
\begin{itemize}[noitemsep]
\item \textbf{I1 Non-ruin}: SAFE\_EXIT / HOLD must not cross
$P(\mathrm{Ruin}) > \varepsilon$.
\item \textbf{I2 Type ZERO mode constraints respected}: action
within $A_{\mathrm{allowed}}$.
\item \textbf{I3 Shutdown atomicity}: if triggered, no partial
execution.
\item \textbf{I4 APCSO path}: PP $\to$ APCSO $\to$ NRMO; no bypass.
\item \textbf{I5 No personality assigned to modules}.
\end{itemize}
}
