\section{Daily Mode (Long-Term Layer Dominant)}
\label{sec:integrated-flow-daily}

\begin{lstlisting}[language=Python,basicstyle=\ttfamily\small,frame=single]
while system_active:
    outer_shell.observe()
    if outer_shell.continuity_maintained():
        continue
    if outer_shell.rupture_detected():
        if rupture.source == "affective":
            humidity_buffer.hold_state()
            continue
        if rupture.source == "irreversible_risk":
            goto HIGH_SPEED_MODE
\end{lstlisting}

\section{Fast Mode (Short-Term Layer Activation)}
\label{sec:integrated-flow-fast}

\begin{lstlisting}[language=Python,basicstyle=\ttfamily\small,frame=single]
HIGH_SPEED_MODE:
    # Kill-Switch evaluation
    for assumption in ASSUMPTIONS:
        if assumption.kill_condition(state):
            action_space = RESTRICTED
            break

    # Abort-First
    if abort_signal_detected():
        return ABORT_ACTION

    # Pre-Commit check
    if pre_commit_triggered(action):
        lock_commitment(action, expiry=T)

    # APCSO proposal generation
    proposals = apcso.generate(state, epsilon)
    # => exactly 3 proposals, >= 1 HOLD/EXIT

    # NRMO Core veto
    selected = human.select(proposals)
    if nrmo_core.veto(selected):
        return SAFE_EXIT

    # Execute
    return execute(selected)
\end{lstlisting}
