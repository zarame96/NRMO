\section{Execution Order}
\label{sec:v21-execution-order}

\begin{enumerate}
\item \texttt{ReflexLayer.evaluate()} $\to$ \texttt{HoldImmediate}?
$\to$ exit.
\item \texttt{ModeSelector.current\_mode()}.
\item \texttt{NonErgodicMonitor.evaluate()} $\to$
\texttt{SafeExit}? $\to$ exit.
\item \texttt{MetaGovernanceLayer.pre\_check(dsi)}:
\begin{itemize}
\item CrossDomainRuin $\to$ \texttt{SafeExit}? $\to$ force
Institutional, exit.
\item DSI monitoring $\to$ \texttt{ForceInstitutional}? $\to$ demote
mode.
\end{itemize}
\item \texttt{TimeHorizonLayer.evaluate()} $\to$ set
\texttt{warn\_flag} (display only).
\item \texttt{NRMOSystemV20.process()} \textbf{[FROZEN v2.0
pipeline]}.
\item Output: \{mode, time\_horizon\_warning, verdicts\}.
\end{enumerate}
